function deferred<T>() {
  let resolve!: (value: T) => void;
  const promise = new Promise<T>(done => { resolve = done; });
  return { promise, resolve };
}
const capabilities = { move: true, resize: true, reshape: true, contents: true, appearance: true, remove: true, reply: true };
function textAnnotation(): PdfFreeTextAnnotation {
  return { id: 'native-text', kind: 'freeText', pageId: 'first', rect: [40, 300, 240, 350], contents: '原生中文', author: '作者',
    capabilities, color: [0.6, 0.4, 0.2], fill: null, opacity: 0.7,
    stroke: { width: 0, style: 'S', lineCap: 0, lineJoin: 0, miterLimit: 10 },
    font: { family: 'Noto Sans SC', size: 16, color: [0.1, 0.2, 0.3] }, align: 0, rectDifferences: [0, 0, 0, 0], lineHeight: 19.2,
    native: { key: '12 0 R', subtype: 'FreeText', flags: 4, baselineHash: 'a'.repeat(64), hasAppearance: true,
      sourceColors: { stroke: { space: 'cmyk', components: [0, 0.2, 0.4, 0.4] } } },
  };
}
function initial(): StudioPdfSnapshot {
  return { source: new Blob(['%PDF-Original']), document: { version: 1, kind: 'pdf', id: 'source', name: 'Source.pdf',
    pageCount: 2, sourcePageCount: 2, pages: [{ id: 'first', sourcePage: 0 }, { id: 'second', sourcePage: 1 }],
    marks: [], fields: [{ name: 'name', type: 'text', value: 'Original', readOnly: false, widgets: [{ page: 0, x: 20, y: 20, width: 160, height: 24 }] }], rotations: {},
    nativeAnnotations: { version: 1, sourceHash: 'a'.repeat(64), bindings: [0, 1].map(sourcePage => ({ sourcePage, nativePageId: `page-${sourcePage}`, uiPageId: sourcePage ? 'second' : 'first' })), commands: [] },
  } };
}
function host(options: { office?: boolean; provider?: boolean; callback?: boolean; signed?: boolean; apply?: StudioPdfAnnotationEditor['apply']; inspect?: StudioPdfAnnotationEditor['inspect']; preview?: StudioPdfAnnotationEditor['preview']; search?: StudioPdfSession['search']; model?: StudioPdfSnapshot } = {}) {
  const first = options.model ?? initial();
  const annotations = new Map<Blob, PdfAnnotation[]>([[first.source, [textAnnotation()]]]);
  const counts = new Map<Blob, number>([[first.source, first.document.pageCount]]);
  let latest = first, accepted = true, sequence = 0;
  const inspection = (snapshot: StudioPdfSnapshot): StudioPdfAnnotationInspection => ({
    state: snapshot.document.nativeAnnotations!, snapshot: { version: 1, sourceHash: snapshot.document.nativeAnnotations!.sourceHash, signed: options.signed ?? false,
      annotations: annotations.get(snapshot.source) ?? [textAnnotation()], deleted: [], removedLinks: [], removedPopups: [],
      pages: snapshot.document.pages!.map(page => ({ id: page.id, sourcePage: page.sourcePage,
        media: { x: 0, y: 0, width: 400, height: 500 }, crop: { x: 0, y: 0, width: 400, height: 500 }, rotation: 0, userUnit: 1 })),
      canvasPages: snapshot.document.pages!, effects: { removedReplies: [], removedLinks: [], removedPopups: [] },
    },
  });
  const candidate = (request: StudioPdfAnnotationRequest): StudioPdfSnapshot => {
    const original = request.snapshot.document, next = request.next ?? original;
    const source = new Blob([`%PDF-Prepared-${++sequence}`]);
    const document: StudioPdfDocument = { ...next, sourcePageCount: next.pageCount, rotations: {},
      pages: next.pages!.map((page, sourcePage) => ({ ...page, sourcePage })),
      nativeAnnotations: { version: 1, sourceHash: String(sequence).padStart(64, '0'), commands: [],
        bindings: next.pages!.map((page, sourcePage) => ({ sourcePage, nativePageId: `native-${page.id}`, uiPageId: page.id })) },
    };
    // The host fixture supplies a completed immutable candidate; native serialization is tested by the real host suites.
    let projected = annotations.get(request.snapshot.source) ?? [textAnnotation()];
    if (request.command.type === 'mark') {
      const command = request.command.command;
      if (command.type === 'update') projected = projected.map(item => item.id === command.id ? { ...item, ...command.patch } as PdfAnnotation : item);
      if (command.type === 'remove') projected = projected.filter(item => item.id !== command.id);
      if (command.type === 'create') projected = [...projected, { ...textAnnotation(), ...command.annotation, capabilities } as PdfAnnotation];
    }
    annotations.set(source, projected); counts.set(source, document.pageCount);
    expect(isOfficeStudioDocument(document)).toBe(true);
    return { source, document };
  };
  const editor: StudioPdfAnnotationEditor = {
    inspect: vi.fn(async (snapshot, settings) => options.inspect ? options.inspect(snapshot, settings) : inspection(snapshot)),
    apply: vi.fn(async request => options.apply ? options.apply(request) : candidate(request)),
    preview: vi.fn(async (snapshot, settings) => options.preview ? options.preview(snapshot, settings) : ({ source: snapshot.source, revision: snapshot.document.nativeAnnotations!.sourceHash,
      rendersCurrentPageOrder: true, nativeBodyIds: [], noApIconIds: [], requiresAnnotationLayer: true,
      binding: { rendersCurrentPageOrder: true, pageCount: snapshot.document.pageCount, sourcePageCount: snapshot.document.sourcePageCount!,
        pages: snapshot.document.pages!.map((page, sourcePage) => ({ id: page.id, sourcePage })), rotations: {} },
    })),
  };
  const renderPage = vi.fn<StudioPdfSession['render']>(async (_page, _canvas, settings) => ({ width: 400 * settings.scale, height: 500 * settings.scale,
    toPdfPoint: (x, y) => [x / settings.scale, 500 - y / settings.scale], toViewportPoint: (x, y) => [x * settings.scale, (500 - y) * settings.scale],
  }));
  const renderer: StudioPdfRenderer = { load: vi.fn(async blob => ({ pageCount: counts.get(blob) ?? 2, supportsPageMapping: true, destroy: vi.fn(),
    render: renderPage, search: options.search,
  })) };
  const modelChange = vi.fn(), sourceChange = vi.fn();
  let update!: (next: StudioPdfSnapshot) => void;
  let permissions!: (next: { readOnly: boolean; disabled: boolean }) => void;
  let replaceEditor!: (next: StudioPdfAnnotationEditor) => void;
  function Host() {
    const [snapshot, setSnapshot] = useState(first), [access, setAccess] = useState({ readOnly: false, disabled: false });
    const [currentEditor, setEditor] = useState(editor);
    latest = snapshot; update = setSnapshot; permissions = setAccess; replaceEditor = setEditor;
    const props = { source: snapshot.source, pdfRenderer: renderer, pdfAnnotations: options.provider === false ? undefined : currentEditor,
      onChange: (document: StudioPdfDocument) => { modelChange(document); setSnapshot(previous => ({ ...previous, document })); },
      onPdfSourceChange: options.callback === false ? undefined : (change: { previous: StudioPdfSnapshot; next: StudioPdfSnapshot }) => {
        sourceChange(change); if (!accepted) return false;
        expect(change.previous.document).toBe(snapshot.document); expect(change.previous.source).toBe(snapshot.source);
        setSnapshot(change.next); return true;
      }, ...access,
    };
    return options.office ? <OfficeStudio value={snapshot.document} {...props} onChange={document => { if (document.kind === 'pdf') props.onChange(document); }} />
      : <StudioPdf document={snapshot.document} {...props} />;
  }
  render(<Host />);
  return { first, editor, renderer, renderPage, candidate, inspection, snapshot: () => latest, modelChange, sourceChange,
    update, permissions, replaceEditor, reject: (value: boolean) => { accepted = !value; } };
}
function expectPair(actual: StudioPdfSnapshot, expected: StudioPdfSnapshot) {
  expect(actual.document).toBe(expected.document); expect(actual.source).toBe(expected.source);
}

function pendingSnapshot() {
  const value = initial();
  value.document.pages = [value.document.pages![1], value.document.pages![0]];
  value.document.rotations = { '0': 90 };
  value.document.nativeAnnotations!.commands = [{ type: 'layout', targets: [{ id: 'second', fromPageId: 'second' }, { id: 'first', fromPageId: 'first' }] },
    { type: 'mark', command: { type: 'update', id: 'native-text', patch: { kind: 'freeText', contents: '重放内容' }, now: '2026-11-27T00:00:00.000Z' } }];
  return value;
}
function previewCandidate(snapshot: StudioPdfSnapshot) {
  return { source: new Blob(['%PDF-Replayed-View']), revision: 'b'.repeat(64), rendersCurrentPageOrder: true as const,
    nativeBodyIds: ['native-text'], noApIconIds: [], requiresAnnotationLayer: true,
    annotations: [{ id: 'native-text', key: '90 0 R', flags: 4 }],
    binding: { rendersCurrentPageOrder: true as const, pageCount: snapshot.document.pageCount, sourcePageCount: snapshot.document.pageCount,
      pages: snapshot.document.pages!.map((entry, sourcePage) => ({ id: entry.id, sourcePage })), rotations: {} },
  };
}

    const value = f.candidate(request);

    if (fault === 'document-id') value.document.id = 'other-file';
    if (fault === 'page-map') value.document.pages![0].sourcePage = 1;
    if (fault === 'recipe') value.document.nativeAnnotations!.sourceHash = 'bad';
    return value;
  } });
  await ready(); const dialog = await properties(); fireEvent.change(within(dialog).getByLabelText('PDF 批注内容'), { target: { value: 'Rejected candidate' } });

});
