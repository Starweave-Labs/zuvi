async function fixture(type = 'text') {
  const pdf = await PDFDocument.create(), font = await pdf.embedFont(StandardFonts.Helvetica), page = pdf.addPage([400, 500]);
  let field;
  if (type === 'radio') {
    field = pdf.getForm().createRadioGroup('Shared'); field.addOptionToPage('A', page, { x: 20, y: 310, width: 20, height: 20 });

    field.acroField.dict.set(n('DV'), field.acroField.getWidgets()[0].getOnValue());
  } else {
    field = pdf.getForm().createTextField('Shared'); field.setText('Parent-current');
    field.addToPage(page, { x: 20, y: 300, width: 180, height: 24, font });
    field.acroField.dict.set(n('DV'), PDFString.of('Parent-default'));
    if (type === 'password') field.enablePassword();
  }
  if (type === 'radio') field.updateAppearances(); else field.updateAppearances(font);
  field.acroField.dict.set(n('DA'), PDFString.of('/F1 12 Tf 0 g'));

  return { pdf, page, font, field, other };
}
function hierarchy({ pdf, field }) {
  const parent = pdf.context.obj({ T: PDFString.of('Root'), Kids: [field.ref] });
  for (const key of ['FT', 'Ff', 'V', 'DV', 'DA', 'Q', 'MaxLen', 'Opt']) {
    const value = field.acroField.dict.get(n(key));
    if (value !== undefined) parent.set(n(key), value);
    field.acroField.dict.delete(n(key));
  }
  const parentRef = pdf.context.register(parent); field.acroField.dict.set(n('Parent'), parentRef);
  const roots = pdf.getForm().acroForm.Fields(), index = roots.indexOf(field.ref); roots.set(index, parentRef);
  return { parent, parentRef };
}
function combine({ pdf, field, page }) {
  const widget = field.acroField.getWidgets()[0], widgetRef = field.acroField.Kids().get(0);
  for (const [key, value] of widget.dict.entries()) if (key !== n('Parent')) field.acroField.dict.set(key, value);
  field.acroField.dict.delete(n('Kids'));
  page.node.Annots().set(page.node.Annots().indexOf(widgetRef), field.ref); pdf.context.delete(widgetRef);
  return field.acroField.dict;
}

  ['widget T', f => f.field.acroField.getWidgets()[0].dict, 'T'],
  ['widget FT', f => f.field.acroField.getWidgets()[0].dict, 'FT'],
  ['combined Kids', combine, 'Kids'],
  ['combined root Parent', combine, 'Parent'],
  ['inherited text FT', f => { hierarchy(f); return f.field.acroField.dict; }, 'FT'],
  ['inherited text V', f => { hierarchy(f); return f.field.acroField.dict; }, 'V'],
  ['inherited password Ff', f => { hierarchy(f); return f.field.acroField.dict; }, 'Ff', 'password'],
